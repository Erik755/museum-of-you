    if (statusEl) statusEl.textContent = "No hay espacio en el teléfono para más fotos.";
  }
}

async function saveDraft(image) {
  try { await idbSet("draft", image || ""); } catch (_) {}
}
async function clearDraft() {
  try { await idbDel("draft"); } catch (_) {}
}
async function restoreDraft() {
  try {
    const draft = await idbGet("draft");
    if (draft && typeof draft === "string" && draft.indexOf("data:") === 0) {
      applyPreview(draft, "Borrador de foto restaurado.");
    }
  } catch (_) {}
}

function applyPreview(dataUrl, msg) {
  currentImage = dataUrl;
  if (previewEl) {
    previewEl.src = dataUrl;
    previewEl.hidden = false;
  }
  if (generateEl) generateEl.disabled = false;
  if (statusEl && msg) statusEl.textContent = msg;
}

async function persistSetup() {
  if (!apiKeyEl) return { lsOk: false, idbOk: false };
  const key = apiKeyEl.value.trim();
  const provider = currentProvider();
  const modelVal = modelEl ? String(modelEl.value || "") : "";
  const ownerVal = ownerEl ? ownerEl.value.trim() : "";
  const promptVal = promptEl ? (promptEl.value || DEFAULT_PROMPT) : DEFAULT_PROMPT;
  let lsOk = true;

  if (!ls(PROVIDER, provider)) lsOk = false;
  if (key) {
    if (!ls(keyStore(), key)) lsOk = false;
  }
  // Always persist model/owner/prompt (even empty model string) so restore can re-select
  if (!ls(MODEL, modelVal)) lsOk = false;
  if (!ls(OWNER, ownerVal)) lsOk = false;
  if (!ls(PROMPT, promptVal)) lsOk = false;

  let existing = null;
  try { existing = await idbGet("settings"); } catch (_) {}
  if (!existing || typeof existing !== "object") existing = {};

  const settings = {
    provider,
    [KEY_GROQ]: provider === "groq" && key ? key : (ls(KEY_GROQ) || existing[KEY_GROQ] || ""),
    [KEY_GEM]: provider === "gemini" && key ? key : (ls(KEY_GEM) || existing[KEY_GEM] || ""),
    model: modelVal || ls(MODEL) || existing.model || "",
    owner: ownerVal,
    prompt: promptVal
  };
  // If ls failed for keys, still keep typed key in IDB
  if (provider === "groq" && key) settings[KEY_GROQ] = key;
  if (provider === "gemini" && key) settings[KEY_GEM] = key;

  let idbOk = false;
  try {
    await idbSet("settings", settings);
    idbOk = true;
  } catch (_) {
    idbOk = false;
  }
  return { lsOk, idbOk };
}

function setModelValue(want) {
  if (!modelEl || !want) return false;
  const opts = [...modelEl.options];
  const hit = opts.find((o) => o.value === want);
  if (hit) {
    modelEl.value = want;
    return modelEl.value === want;
  }
  // Keep custom/saved model visible even if not in the static list
  const opt = document.createElement("option");
  opt.value = want;
  opt.textContent = want;
  modelEl.appendChild(opt);
  modelEl.value = want;
  return true;
}

async function restoreSetup() {
  let idbSettings = null;
  try { idbSettings = await idbGet("settings"); } catch (_) {}
  if (!idbSettings || typeof idbSettings !== "object") idbSettings = null;

  const pick = (lsKey, idbField) => {
    const fromLs = ls(lsKey);
    if (fromLs != null && fromLs !== "") return fromLs;
    if (idbSettings && idbSettings[idbField] != null && String(idbSettings[idbField]) !== "") {
      return String(idbSettings[idbField]);
    }
    // allow empty string from ls if key exists
    if (fromLs != null) return fromLs;
    return "";
  };

  const old = ls("moy_groq_key") || pick(KEY_GROQ, KEY_GROQ) || "";
  const provider = pick(PROVIDER, "provider") || (pick(KEY_GEM, KEY_GEM) ? "gemini" : "groq");
  if (providerEl) providerEl.value = provider === "gemini" ? "gemini" : "groq";

  const wantModel = pick(MODEL, "model");
  if (wantModel) setModelValue(wantModel);

  if (apiKeyEl) {
    const store = provider === "gemini" ? KEY_GEM : KEY_GROQ;
    apiKeyEl.value = pick(store, store) || (provider === "groq" ? old : "") || "";
    apiKeyEl.placeholder = provider === "gemini" ? "AIzaSy..." : "gsk_...";
  }
  if (ownerEl) ownerEl.value = pick(OWNER, "owner") || "";
  if (promptEl) promptEl.value = pick(PROMPT, "prompt") || DEFAULT_PROMPT;

  // Rehydrate localStorage from IDB when possible (quota may still fail)
  if (idbSettings) {
    if (!ls(PROVIDER)) ls(PROVIDER, provider);
    if (idbSettings[KEY_GROQ] && !ls(KEY_GROQ)) ls(KEY_GROQ, idbSettings[KEY_GROQ]);
    if (idbSettings[KEY_GEM] && !ls(KEY_GEM)) ls(KEY_GEM, idbSettings[KEY_GEM]);
    if (idbSettings.model && !ls(MODEL)) ls(MODEL, idbSettings.model);
    if (idbSettings.owner != null && !ls(OWNER)) ls(OWNER, idbSettings.owner);
